from skimage.metrics import peak_signal_noise_ratio, structural_similarity
        # CNMF doesn't need training
        cnmf_model = CNMFBaseline()
        models['CNMF_Baseline'] = cnmf_model

        # ===== Validation on supplied Validation/ folder (F0..F3) =====
        print("\nðŸ“Š Running validation on Validation folder (F0..F3) if present...")
        try:
            self.validate_on_validation_set(models, calcium_dataset)
        except Exception as _val_e:
            print(f"âŒ Validation step failed: {_val_e}")

    def validate_on_validation_set(self, models, dataset):
        """Validate trained models on a small held-out Validation set folder.

        Looks for files named F0.tif (ground truth), F1.tif, F2.tif, F3.tif in one of these locations (in order):
          1) a folder named 'Validation' in the current working directory
          2) a folder named 'Validation' sibling to the Train folder (os.path.join(os.path.dirname(dataset_path), '..', 'Validation'))
          3) a folder at the same dir as dataset file: os.path.join(os.path.dirname(self.dataset_path), 'Validation')

        Produces per-frame MSE/PSNR/SSIM and saves denoised stacks and a CSV report in output_dir/validation_results.
        """
        import os
        from skimage.metrics import peak_signal_noise_ratio, structural_similarity
        import tifffile
        import pandas as pd
        import matplotlib.pyplot as plt

        # Build candidate validation paths
        cand_paths = [
            os.path.join(os.getcwd(), 'Validation'),
            os.path.join(os.path.dirname(self.dataset_path), '..', 'Validation'),
            os.path.join(os.path.dirname(self.dataset_path), 'Validation')
        ]

        validation_dir = None
        for p in cand_paths:
            if p and os.path.isdir(p):
                validation_dir = p
                break

        if validation_dir is None:
            print('âŒ No Validation directory found in expected locations. Skipping validation step.')
            return

        print(f"ðŸ“Š Found Validation directory: {validation_dir}")

        # Expected files
        gt_name = 'F0.tif'
        noisy_names = ['F1.tif','F2.tif','F3.tif']

        gt_path = os.path.join(validation_dir, gt_name)
        if not os.path.isfile(gt_path):
            print(f"âŒ Ground truth file {gt_name} not found in {validation_dir}. Skipping validation.")
            return

        gt_stack = tifffile.imread(gt_path).astype(np.float32)

        # Normalize ground truth using robust percentile scaling (1-99)
        p1, p99 = np.percentile(gt_stack, [1,99])
        gt_stack = np.clip(gt_stack, p1, p99)
        gt_stack = (gt_stack - p1) / (p99 - p1 + 1e-12)

        # Create output folder
        val_out = os.path.join(self.output_dir, 'validation_results')
        os.makedirs(val_out, exist_ok=True)

        report_rows = []

        for noisy_file in noisy_names:
            noisy_path = os.path.join(validation_dir, noisy_file)
            if not os.path.isfile(noisy_path):
                print(f"âŒ Missing {noisy_file} in Validation folder, skipping this file.")
                continue

            noisy_stack = tifffile.imread(noisy_path).astype(np.float32)
            # Normalize using same robust scaling
            noisy_stack = np.clip(noisy_stack, p1, p99)
            noisy_stack = (noisy_stack - p1) / (p99 - p1 + 1e-12)

            print(f"ðŸ“Š Validating using {noisy_file} (shape {noisy_stack.shape})")

            for model_name, model in models.items():
                # Skip untrained CNMF baseline for direct model inference if it's not a torch Module
                is_torch_module = isinstance(model, torch.nn.Module)

                denoised_stack = None
                try:
                    if is_torch_module:
                        model.eval()
                        with torch.no_grad():
                            # Prepare input as [B, T, C, H, W]
                            T, H, W = noisy_stack.shape
                            inp = torch.from_numpy(noisy_stack).unsqueeze(0).unsqueeze(2).float().to(self.device)

                            out = model(inp)
                            out_np = out.squeeze(0).squeeze(1).cpu().numpy()
                            denoised_stack = out_np
                    else:
                        # CNMF/NMF path: use fit_denoise if available
                        if hasattr(model, 'fit_denoise'):
                            denoised_stack = model.fit_denoise(noisy_stack)
                        else:
                            print(f"âŒ Model {model_name} is not runnable for inference. Skipping.")
                            continue

                    # Clip and rescale to [0,1]
                    denoised_stack = np.clip(denoised_stack, 0, 1)

                    # Compute per-frame metrics
                    frame_metrics = []
                    n_frames = min(denoised_stack.shape[0], gt_stack.shape[0])
                    for f in range(n_frames):
                        pred_f = denoised_stack[f]
                        gt_f = gt_stack[f]
                        mse_f = np.mean((pred_f - gt_f)**2)
                        psnr_f = peak_signal_noise_ratio(gt_f, pred_f, data_range=1.0)
                        try:
                            ssim_f = structural_similarity(gt_f, pred_f, data_range=1.0)
                        except Exception:
                            ssim_f = np.nan
                        frame_metrics.append({'frame': f, 'mse': float(mse_f), 'psnr': float(psnr_f), 'ssim': float(ssim_f)})

                    # Aggregate
                    mse_mean = float(np.mean([m['mse'] for m in frame_metrics]))
                    psnr_mean = float(np.mean([m['psnr'] for m in frame_metrics]))
                    ssim_mean = float(np.nanmean([m['ssim'] for m in frame_metrics]))

                    report_rows.append({
                        'noisy_file': noisy_file,
                        'model': model_name,
                        'mse_mean': mse_mean,
                        'psnr_mean': psnr_mean,
                        'ssim_mean': ssim_mean,
                        'n_frames': n_frames
                    })

                    # Save denoised stack as tiff
                    out_path = os.path.join(val_out, f"{model_name}_{noisy_file.replace('.tif','')}_denoised.tif")
                    tifffile.imwrite(out_path, (denoised_stack * 65535).astype(np.uint16))

                    # Save an example figure comparing frame 0 (noisy / denoised / gt)
                    fig, axes = plt.subplots(1,3, figsize=(12,4))
                    axes[0].imshow(noisy_stack[0], cmap='gray')
                    axes[0].set_title('Noisy (frame 0)')
                    axes[1].imshow(denoised_stack[0], cmap='gray')
                    axes[1].set_title(f'Denoised: {model_name}')
                    axes[2].imshow(gt_stack[0], cmap='gray')
                    axes[2].set_title('Ground truth (F0 frame 0)')
                    for ax in axes:
                        ax.axis('off')
                    fig_path = os.path.join(val_out, f"{model_name}_{noisy_file.replace('.tif','')}_example.png")
                    fig.savefig(fig_path, bbox_inches='tight')
                    plt.close(fig)

                    print(f"âœ… Saved denoised tiff and example figure for {model_name} on {noisy_file}")

                except Exception as e:
                    print(f"âŒ Error running inference for {model_name} on {noisy_file}: {e}")
                    continue

        # Save CSV report
        if report_rows:
            df = pd.DataFrame(report_rows)
            csv_path = os.path.join(val_out, 'validation_summary.csv')
            df.to_csv(csv_path, index=False)
            print(f"ðŸ“Š Validation summary saved: {csv_path}")
        else:
            print('âŒ No validation results to save.')
